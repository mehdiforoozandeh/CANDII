% ------------------------------------------------------------------ the words --
\newcommand{\PosterTitle}{Self-supervised confidence-aware denoising imputation of genomic data}
\newcommand{\PosterAuthors}{Mehdi Foroozandeh, Abdul Rahman Diab, and Maxwell Libbrecht}
\newcommand{\PosterURL}{https://mehdiforoozandeh.github.io/CANDI/}

\newcommand{\TxtProblem}{%
Epigenomic measurements often suffer from noise, batch effects and
irreproducibility. Imputation methods operate on processed signal, and assume that
processing removes all batch effects. In the ENCODE Imputation Challenge, due to
subtle train-test differences, a simple baseline outperformed most entrants~[1].}

\newcommand{\TxtCandi}{%
\textbf{C}onfidence-\textbf{A}ware \textbf{N}eural \textbf{D}enoising
\textbf{I}mputer~[2] is a self-supervised deep learning method for epigenome
imputation and denoising. It takes raw read counts, DNA sequence and experimental
covariates, and outputs probability distributions over counts, signal and peaks.
Conditioning on metadata rather than cell-type embeddings enables zero-shot
generalization to unseen cell types.}

\newcommand{\TxtFindings}{%
\begin{itemize}
  \item CANDI ranks \textbf{1st of 26} on the ENCODE Imputation Challenge.
  \item CANDI provides calibrated uncertainty estimates.
  \item CANDI's imputed signals and latent space predict gene expression.
\end{itemize}}

\newcommand{\HeadStrip}{CANDI schematic workflow}
\newcommand{\CapStrip}{%
\textbf{Top:} for each 30\,kb locus, CANDI takes observed read counts with the
ChIP-seq control, four covariates per assay and the DNA sequence; grey assays were
never measured, yellow ones are masked. \textbf{Bottom:} each output is a
probability distribution, here a mean and 95\% interval per assay; window by
window, they fill every missing thread of the tensor.}

\newcommand{\HeadUtility}{Downstream utility of CANDI}
\newcommand{\CapA}{%
\textbf{A}\enspace Confidence calibration for DNase-seq (DND-41): predicted
confidence level against the fraction of observed data within that interval.}
\newcommand{\CapB}{%
\textbf{B}\enspace Gene expression prediction as input assays are removed.
Denoised+Imputed and Latent settings are robust to input sparsity.}

\newcommand{\HeadEIC}{Fixing the signal-level shift in the ENCODE Imputation Challenge blind test}
\newcommand{\TxtEIC}{%
\textbf{The shift.} We followed the EIC train-test split: 51 blind-test
experiments (12 cell types, 8 assays). The train and test sets differ subtly:
every blind-test experiment is paired-end, against 15\% of training experiments,
and blind-test tracks sit at an assay-specific signal level. CANDI recovers the
rank-ordering and structure of the signal but not its absolute magnitude; its raw
output ranks 14th.
\par\vspace{0.2in}
\textbf{The fix.} Per assay, CANDI's output passes through one layer,
\par\vspace{0.12in}
\centerline{$\max(\beta_0 + \beta_1\cdot\text{CANDI} + \beta_2\cdot\text{baseline},\,0)$,}
\vspace{0.12in}
fitted by least squares on the 133 training experiments only. CANDI then ranks
1st.}
\newcommand{\CapC}{%
\textbf{C}\enspace The leaderboard with CANDI added: CANDI is first at 0.179
(bootstrap range 0.177--0.184), the next entry 0.306.}
\newcommand{\CapD}{%
\textbf{D}\enspace Eight of the nine measures (top-1\%-imputed MSE not shown)
relative to the baseline, median over 51 experiments. CANDI is first on three and
top 8 on the rest.}

\newcommand{\TxtRefs}{%
[1]~Schreiber J, Boix C, \textit{et al.} The ENCODE Imputation Challenge: a critical
assessment of methods for cross-cell type imputation of epigenomic profiles.
\textit{Genome Biol} 24:79 (2023).\par\vspace{0.1in}
[2]~Foroozandeh M, Diab AR, Libbrecht M. CANDI: self-supervised, confidence-aware
denoising imputation of genomic data. \textit{bioRxiv} 2025.01.23.634626 (2025).\par
\vspace{0.1in}
Panels C--D: blind-test scoring from the Libbrecht lab's EIC reproduction
(experiments 017, 025), which reproduces every published entry to a median
relative difference of $9\times10^{-10}$; \emph{baseline} is the challenge's
average-activity track.
The correction was chosen after the challenge results were public; none of it is
fitted on blind-test data.}
